    <script>
        // Инициализация Telegram Web App
        const tg = window.Telegram.WebApp;
        tg.expand();

        // Подставляем имя и дефолтный баланс (позже свяжем с базой данных)
        let userBalance = 100;
        if (tg.initDataUnsafe && tg.initDataUnsafe.user) {
            document.getElementById('username').innerText = tg.initDataUnsafe.user.first_name;
        }

        function updateBalanceUI() {
            document.getElementById('user-balance').innerText = userBalance;
        }

        // Логика анимации открытия кейса
        function openCase(caseName, cost) {
            if (userBalance < cost) {
                alert('❌ Недостаточно звёзд на балансе!');
                return;
            }

            userBalance -= cost;
            updateBalanceUI();

            const modal = document.getElementById('modal');
            const heading = document.getElementById('modal-heading');
            const rouletteText = document.getElementById('roulette-text');
            const closeBtn = document.getElementById('modal-close-btn');

            modal.style.display = 'flex';
            heading.innerText = `Открываем ${caseName}...`;
            closeBtn.style.display = 'none';

            // Имитация крутящейся рулетки
            let counter = 0;
            const items = ['+3 ⭐', '+5 ⭐', '+10 ⭐', '🔥 ДЖЕКПОТ 50 ⭐', '+2 ⭐', '💎 NFT-Гифт'];
            
            const spinInterval = setInterval(() => {
                rouletteText.innerText = items[Math.floor(Math.random() * items.length)];
                counter++;
                
                if (counter > 15) {
                    clearInterval(spinInterval);
                    // Финальный выигрыш
                    const win = items[Math.floor(Math.random() * items.length)];
                    rouletteText.innerText = `🎉 Выиграно:\n${win}`;
                    heading.innerText = "Поздравляем!";
                    closeBtn.style.display = 'block';
                }
            }, 100);
        }

        function closeModal() {
            document.getElementById('modal').style.display = 'none';
        }
    </script>
